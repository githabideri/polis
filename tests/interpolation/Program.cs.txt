using System;
using System.Collections.Generic;
using System.Reflection;
using System.Runtime.CompilerServices;
using HarmonyLib;
using Vintagestory.API.Client;
using Vintagestory.API.Common;
using Vintagestory.API.Common.Entities;

// Supply only the client interfaces touched by interpolation. All arithmetic,
// queue processing, and position writes below are the installed engine's code.
public class ClientStub : DispatchProxy
{
    protected override object Invoke(MethodInfo method, object[] args)
    {
        if (method.Name == "get_IsGamePaused") return false;
        if (method.Name == "get_World") return Create<IClientWorldAccessor, ClientStub>();
        if (method.Name == "get_Player")
        {
            var player = RuntimeHelpers.GetUninitializedObject(typeof(Vintagestory.Client.NoObf.ClientPlayer));
            var field = AccessTools.Field(player.GetType(), "worlddata");
            field.SetValue(player, RuntimeHelpers.GetUninitializedObject(field.FieldType));
            return player;
        }
        if (method.Name == "get_Entity") return null; // local player differs from tested entity
        throw new NotSupportedException(method.ToString());
    }
}
class TestEntity : Entity {}
class Program
{
    static EntityBehaviorInterpolatePosition NewBehavior()
    {
        var b = (EntityBehaviorInterpolatePosition)RuntimeHelpers.GetUninitializedObject(typeof(EntityBehaviorInterpolatePosition));
        b.entity = new TestEntity();
        b.entity.Pos.SetPos(100, 50, 100);
        b.capi = DispatchProxy.Create<ICoreClientAPI, ClientStub>();
        b.positionQueue = new Queue<PositionSnapshot>();
        b.pL = b.pN = new PositionSnapshot(b.entity.Pos, 1f / 15f, false);
        b.targetSpeed = 0.6f;
        return b;
    }
    static int Run(float[] frameTimes, int frames, bool moving)
    {
        var b = NewBehavior();
        for (int i = 0; i < frames; i++)
        {
            // Model a finite server packet arriving before each render.
            b.entity.Pos.SetPos(100 + (moving ? Math.Sin(i * .03) : 0), 50, 100);
            b.entity.Attributes.SetInt("tickDiff", 1);
            var handling = EnumHandling.PassThrough;
            b.OnReceivedServerPos(false, ref handling);
            b.OnRenderFrame(frameTimes[i % frameTimes.Length], EnumRenderStage.Before);
            var p = b.entity.Pos;
            if (!double.IsFinite(p.X) || !double.IsFinite(p.Y) || !double.IsFinite(p.Z)) return i + 1;
        }
        return 0;
    }
    static void Require(bool ok, string message)
    {
        if (!ok) throw new Exception(message);
    }
    static void Main()
    {
        int failed = Run(new[] { 1f }, 2000, false);
        Require(failed > 0, "Original engine did not reproduce the failure.");
        Console.WriteLine("Original engine: non-finite position at frame " + failed);
        var harmony = new Harmony("polis.interpolation.regression");
        var method = AccessTools.Method(typeof(EntityBehaviorInterpolatePosition), "OnRenderFrame");
        harmony.Patch(method, transpiler: new HarmonyMethod(typeof(PolisInterpolationStability), "Transpiler"));
        float[][] schedules = {
            new[] { 1f / 60f }, new[] { 1f / 15f }, new[] { .5f },
            new[] { 2f / 3f }, new[] { 1f }, new[] { 5f },
            new[] { 1f / 60f, 1f, .1f, 3f, .6f, 0f, 10f }
        };
        foreach (var schedule in schedules)
        {
            foreach (bool moving in new[] { false, true })
                Require(Run(schedule, 20000, moving) == 0, "Patched engine produced a non-finite position.");
        }
        // Fast frames preserve the engine's exact float arithmetic.
        foreach (float amount in new[] { 0f, .01f, .2f, .99f, 1f })
            Require(PolisInterpolationStability.StableSpeedLerp(.6f, 2f, amount) ==
                Vintagestory.API.MathTools.GameMath.Lerp(.6f, 2f, amount), "Normal-frame behavior changed.");
        Console.WriteLine("Patched engine: 280000 frames PASS (stationary/moving, 60 to 0.2 FPS, stalls/zero dt).");
        Console.WriteLine("Normal-frame speed arithmetic: unchanged.");
        harmony.UnpatchAll("polis.interpolation.regression");
    }
}
